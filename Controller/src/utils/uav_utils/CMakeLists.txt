cmake_minimum_required(VERSION 3.16)
project(uav_utils)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/odom_to_euler.py scripts/tf_assist.py scripts/send_odom.py
  DESTINATION lib/${PROJECT_NAME})
ament_export_include_directories(include ${EIGEN3_INCLUDE_DIRS})
ament_export_dependencies(Eigen3 nav_msgs geometry_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(uav_utils_test src/uav_utils_test.cpp)
  target_compile_features(uav_utils_test PRIVATE cxx_std_17)
  target_include_directories(uav_utils_test PRIVATE include)
  target_link_libraries(uav_utils_test Eigen3::Eigen)
  ament_target_dependencies(uav_utils_test nav_msgs geometry_msgs)
endif()
ament_package()
